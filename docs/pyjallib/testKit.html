pyjallib.testKit

pyjallib.testKit 패키지

DCC 독립적 테스트 자동화 인프라. stdlib만 사용하여 어떤 Python 환경에서든 추가 설치 없이 동작한다.

주요 모듈: - testReporter: DCC 내부에서 사용하는 테스트 리포터 - testLogAnalyzer: 외부에서 로그 파일을 파싱하는 분석기 - testRunner: subprocess로 DCC 프로세스를 실행하는 코어 러너

class TestLogAnalyzer: 
 42class TestLogAnalyzer:
 43    """테스트 로그 파일을 파싱하여 구조화된 결과를 반환하는 분석기.
 44
 45    TestReporter가 생성한 표준 로그 포맷을 파싱한다.
 46    SUCCESS/FAIL/ERROR 패턴과 TEST END 마커를 인식한다.
 47
 48    Example:
 49        analyzer = TestLogAnalyzer()
 50        result = analyzer.analyze(Path("tests/logs/test_CheckLayer.log"))
 51        if analyzer.is_passed(result):
 52            print("모든 테스트 통과")
 53    """
 54
 55    # 로그 라인 패턴 (타임스탬프 뒤의 내용을 매칭)
 56    _SUCCESS_PATTERN: re.Pattern[str] = re.compile(r"SUCCESS: (.+)")
 57    _FAIL_PATTERN: re.Pattern[str] = re.compile(r"FAIL: (.+)")
 58    _ERROR_PATTERN: re.Pattern[str] = re.compile(r"ERROR: (.+)")
 59    _END_PATTERN: re.Pattern[str] = re.compile(r"=== TEST END.*===")
 60
 61    def analyze(self, inLogPath: Path) -> TestResult:
 62        """단일 로그 파일을 파싱하여 TestResult를 반환한다.
 63
 64        Args:
 65            inLogPath: 분석할 로그 파일 경로
 66
 67        Returns:
 68            파싱된 테스트 결과
 69        """
 70        result = TestResult()
 71        logPath = Path(inLogPath)
 72
 73        if not logPath.exists():
 74            return result
 75
 76        content = logPath.read_text(encoding="utf-8")
 77
 78        for line in content.splitlines():
 79            # ERROR를 먼저 검사하여 "FAIL:" 라인에 "ERROR:"가 포함된 경우를
 80            # 올바르게 에러로 분류한다. 순서: ERROR -> FAIL -> SUCCESS -> END
 81            errorMatch = self._ERROR_PATTERN.search(line)
 82            if errorMatch is not None:
 83                result.errors += 1
 84                result.error_details.append(errorMatch.group(1))
 85                continue
 86
 87            failMatch = self._FAIL_PATTERN.search(line)
 88            if failMatch is not None:
 89                result.failed += 1
 90                result.failures.append(failMatch.group(1))
 91                continue
 92
 93            successMatch = self._SUCCESS_PATTERN.search(line)
 94            if successMatch is not None:
 95                result.passed += 1
 96                continue
 97
 98            endMatch = self._END_PATTERN.search(line)
 99            if endMatch is not None:
100                result.completed = True
101
102        result.total = result.passed + result.failed + result.errors
103        return result
104
105    def analyze_multiple(self, inLogPaths: list[Path]) -> list[TestResult]:
106        """여러 로그 파일을 일괄 분석한다.
107
108        Args:
109            inLogPaths: 분석할 로그 파일 경로 목록
110
111        Returns:
112            각 로그 파일에 대한 TestResult 목록
113        """
114        results: list[TestResult] = []
115        for logPath in inLogPaths:
116            results.append(self.analyze(logPath))
117        return results
118
119    def is_passed(self, inResult: TestResult) -> bool:
120        """테스트 결과가 통과인지 판정한다.
121
122        실패(failed)와 에러(errors)가 모두 0이고, 테스트가 정상 완료(completed)되어야
123        통과로 판정한다.
124
125        Args:
126            inResult: 판정할 테스트 결과
127
128        Returns:
129            통과 여부
130        """
131        return (
132            inResult.failed == 0 and inResult.errors == 0 and inResult.completed is True
133        )
134
135    def format_report(self, inResult: TestResult) -> str:
136        """테스트 결과를 사람이 읽기 좋은 텍스트 리포트로 변환한다.
137
138        Args:
139            inResult: 리포트로 변환할 테스트 결과
140
141        Returns:
142            포맷팅된 텍스트 리포트
143        """
144        lines: list[str] = []
145        lines.append("=" * 50)
146
147        statusText = "PASSED" if self.is_passed(inResult) else "FAILED"
148        lines.append(f"테스트 결과: {statusText}")
149        lines.append("-" * 50)
150        lines.append(f"통과: {inResult.passed}")
151        lines.append(f"실패: {inResult.failed}")
152        lines.append(f"에러: {inResult.errors}")
153        lines.append(f"전체: {inResult.total}")
154        lines.append(f"완료: {'예' if inResult.completed else '아니오'}")
155
156        if inResult.failures:
157            lines.append("-" * 50)
158            lines.append("실패 목록:")
159            for failure in inResult.failures:
160                lines.append(f"  - {failure}")
161
162        if inResult.error_details:
163            lines.append("-" * 50)
164            lines.append("에러 목록:")
165            for errorDetail in inResult.error_details:
166                lines.append(f"  - {errorDetail}")
167
168        lines.append("=" * 50)
169        return "\n".join(lines)
