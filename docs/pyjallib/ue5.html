 33    Interchange Framework 기반 템플릿 처리 클래스

 47    def process_template(self, inTemplatePath: str, inTemplateOutPath: str, inTemplateData: Dict[str, Any]) -> str:
 48        """
 49        템플릿을 처리하여 실제 코드로 변환
 50        
 51        Args:
 52            inTemplatePath (str): 템플릿 파일 경로
 53            inTemplateOutPath (str): 출력 파일 경로
 54            inTemplateData (Dict[str, Any]): 템플릿에서 치환할 데이터
 55            
 56        Returns:
 57            str: 처리된 템플릿 내용
 58            
 59        Raises:
 60            FileNotFoundError: 템플릿 파일이 존재하지 않는 경우
 61            PermissionError: 파일 읽기/쓰기 권한이 없는 경우
 62            OSError: 디렉토리 생성 실패 등 파일 시스템 오류
 63            UnicodeDecodeError: 파일 인코딩 오류
 64        """
 65        # 템플릿 파일 존재 확인
 66        templatePath = Path(inTemplatePath)
 67        if not templatePath.exists():
 68            self._logger.error(f"템플릿 파일을 찾을 수 없습니다: {inTemplatePath}")
 69            raise FileNotFoundError(f"템플릿 파일을 찾을 수 없습니다: {inTemplatePath}")
 70        
 71        # 템플릿 파일 읽기 권한 확인
 72        if not os.access(templatePath, os.R_OK):
 73            self._logger.error(f"템플릿 파일 읽기 권한이 없습니다: {inTemplatePath}")
 74            raise PermissionError(f"템플릿 파일 읽기 권한이 없습니다: {inTemplatePath}")
 75        
 76        # 템플릿 파일 읽기
 77        with open(templatePath, 'r', encoding='utf-8') as file:
 78            templateContent = file.read()
 79        
 80        # 템플릿 데이터로 플레이스홀더 치환
 81        processedContent = templateContent
 82        for key, value in inTemplateData.items():
 83            placeholder = f'{{{key}}}'
 84            if placeholder in processedContent:
 85                processedContent = processedContent.replace(placeholder, str(value))
 86        
 87        # 출력 디렉토리 생성 (존재하지 않는 경우)
 88        outputPath = Path(inTemplateOutPath)
 89        outputPath.parent.mkdir(parents=True, exist_ok=True)
 90        
 91        # 출력 파일 쓰기 권한 확인 (기존 파일이 있는 경우)
 92        if outputPath.exists() and not os.access(outputPath, os.W_OK):
 93            self._logger.error(f"출력 파일 쓰기 권한이 없습니다: {inTemplateOutPath}")
 94            raise PermissionError(f"출력 파일 쓰기 권한이 없습니다: {inTemplateOutPath}")
 95        
 96        # 처리된 내용을 파일로 출력
 97        with open(outputPath, 'w', encoding='utf-8') as file:
 98            file.write(processedContent)
 99        
100        self._logger.info(f"템플릿 처리 완료: {inTemplatePath} -> {inTemplateOutPath}")
101        return processedContent
